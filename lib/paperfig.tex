% \paperfig{image.png}{caption.txt}
% Shows a centered image with caption text read from file below it.
% If the caption file does not exist, no caption is shown.
\usepackage{catchfile}

\newcommand{\paperfig}[2]{%
  \begin{center}
  \includegraphics[width=0.85\textwidth,height=0.6\textheight,keepaspectratio]{#1}%
  \end{center}%
  \ifx\relax#2\relax\else
    \IfFileExists{#2}{%
      \CatchFileDef{\captiontext}{#2}{\endlinechar=-1}%
      {\tiny\captiontext}%
    }{}%
  \fi
}


% \fullfig{image.png}{citekey}: figure-only frame, figure fills all space between title and source
\newcommand{\fullfig}[2]{%
  \vspace*{-1ex}%
  \centerline{\includegraphics[width=0.98\paperwidth,height=\dimexpr\textheight-2ex-36pt\relax,keepaspectratio]{#1}}%
  \source{#2}%
}
