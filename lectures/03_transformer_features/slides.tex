\begin{frame}{Getting a Feature Direction}
\begin{columns}[c]
\begin{column}{0.55\textwidth}
\textbf{Contrastive pairs:}
\[ r = \text{Emb}(\text{``Spanish''}) - \text{Emb}(\text{``French''}) \]

\textbf{Linear probe:} \quad {\small $x_i$: activation, $y_i \in \{\pm 1\}$: label}
\[ r = \arg\max_{\|r\|=1} \sum_i y_i \cdot r^\top x_i \]

\textbf{CAA:} \quad {\small $x_i^{(\pm)}$: positive/negative example}
\[ r = \frac{1}{N} \sum_{i=1}^{N} \left( x_i^{(+)} - x_i^{(-)} \right) \]
\end{column}

\note{
\begin{itemize}
\item Preview: three families of methods for obtaining a feature direction
\item Contrastive pairs: just subtract embeddings of paired concepts
\item Linear probe: supervised training on labeled examples
\item CAA: averaged activation differences across many contrastive examples
\item We will see concrete examples of each in the following slides
\end{itemize}
}

\begin{frame}{What To Do with a Feature Direction}
\textbf{Activation addition} (steering):
\[ x' \leftarrow x + \alpha \cdot r \]

\vspace{0.5em}
\textbf{Direction ablation} (make the model blind to the concept):
\[ x' \leftarrow x - \hat{r}\hat{r}^\top x \]

\vspace{0.5em}
\textbf{Probing} (read out a concept):
\[ s = r^\top x \]

\vspace{0.3em}
{\small $r$: feature direction \quad $x$: activation \quad $\alpha$: steering strength}

\note{
\begin{itemize}
\item Once we have a direction, there are three things we can do with it
\item Steering: add it to activations to push the model toward the concept
\item Ablation: project it out to make the model blind to the concept
\item Probing: read off the dot product to detect whether the concept is present
\end{itemize}
}
\end{frame}

\begin{frame}{Refusal Ablation, Baked into the Weights}
\begin{columns}[c]
\begin{column}{0.55\textwidth}
Every matrix that writes to the residual stream:
\[ W_{\text{out}}' \leftarrow W_{\text{out}} - \hat{r}\hat{r}^\top W_{\text{out}} \]
(embedding, attention out, MLP out, biases)

\vspace{0.5em}
\begin{itemize}
\item Equivalent to ablating $\hat r$ at every layer
\item No gradients, no harmful examples
\item Competitive with other general jailbreaks on HarmBench
\end{itemize}
\end{column}
\begin{column}{0.42\textwidth}

\end{columns}

\source{arditi2024}

\note{
\begin{itemize}
\item Under the hood of the ``safety training can be removed'' slide
\item Directional ablation at inference time is equivalent to editing the weights: orthogonalize every matrix that writes into the residual stream against the refusal direction
\item Then the model can never write the refusal direction, so it can never ``notice'' harmfulness in that way
\item Rank-one edit per matrix, no fine-tuning, no gradient steps, no harmful completions needed
\item On HarmBench it is competitive with other general (not prompt-specific) jailbreaks, on Qwen even on par with prompt-specific GCG
\item Coherence mostly preserved, but TruthfulQA drops somewhat
\item For open-weight models this is a cheap way to remove safety training
\end{itemize}
}

\section{Superposition}

\begin{frame}{Johnson--Lindenstrauss Lemma}
In $N$ dimensions, we can pack
\[ m \leq e^{c \cdot \varepsilon^2 \cdot N} \]
vectors with pairwise dot product at most $\varepsilon$.

Features scale \textbf{exponentially} with dimensions, given overlap $\varepsilon$.

\begin{center}
\includegraphics[width=0.85\textwidth]{figures/random_cosines.png}
\end{center}

\source{johnson1984}

\note{
\begin{itemize}
\item The effect is quite big: instead of scaling the number of features linearly with dimensions, we scale exponentially given a feature overlap of $\varepsilon$
\item Plot: two random directions are almost orthogonal in high dimensions. Residual streams have thousands of dimensions
\end{itemize}
}
\end{frame}

\begin{frame}{Dictionary Learning}
\textbf{Setup:} Unknown dictionary $A = [A_1, \ldots, A_m] \in \mathbb{R}^{n \times m}$ with $m > n$.

We observe samples $y = Ax$, where $x \in \mathbb{R}^m$ is $k$-sparse.
We know neither $x$ nor $A$.

\vspace{0.5em}
\textbf{Claim:} $A$ is recoverable up to permutation and sign, in polynomial time, if:
\begin{itemize}
\item \textbf{Sparsity:} $k \leq c \cdot \min\!\left(m^{2/5},\; \frac{\sqrt{n}}{\mu \log n}\right)$
\item \textbf{Incoherence:} $|\langle A_i, A_j \rangle| \leq \frac{\mu}{\sqrt{n}}$ for all $i \neq j$
\item \textbf{Sufficient data:} $p = \Omega\!\left(\frac{m^2}{k^2} \log m\right)$ samples with varying support
\item \textbf{Well-behaved coefficients:} independent, symmetric, bounded away from zero
\end{itemize}

\source{arora2013}

\note{
\begin{itemize}
\item This gives us theoretical backing for why finding features in superposition should even be possible
\item The setup: imagine a set of directions (features), and we observe weighted sums of a few of them at a time
\item We never know which features are active or what the weights are, we just see the resulting vector
\item The claim: under these conditions, the set of directions is uniquely identifiable
\item There is no alternative set of directions and weights that could produce the same observations
\item Sparsity: not too many features active at once
\item Incoherence: the true feature directions are not too similar to each other
\item We need enough data with varying combinations of active features
\item Importantly, this does not tell us that SAEs specifically will find these features
\item It tells us the features are there to be found, and that some polynomial-time algorithm exists
\item Whether gradient descent on the SAE objective converges to the right answer is a separate question
\end{itemize}
}
\end{frame}

\begin{frame}{Compressed Sensing}
\begin{columns}[c]
\begin{column}{0.6\textwidth}
\small
\textbf{Setup:} Known dictionary $A \in \mathbb{R}^{n \times m}$.
We observe $y = Ax$, where $x$ is $k$-sparse.
We want to recover $x$.

\vspace{0.5em}
\textbf{Problem:} $m > n$, so there are infinitely many solutions to $Ax = y$.

\vspace{0.5em}
\textbf{Claim:} If $A$ satisfies the Restricted Isometry Property (RIP), then
\[ \min_z \|z\|_1 \quad \text{s.t.} \quad Az = y \]
recovers $x$ exactly.

\vspace{0.5em}
$\ell_1$ minimization finds the sparse solution; $\ell_2$ minimization does not.
\end{column}
\begin{column}{0.38\textwidth}
\centering
\begin{tikzpicture}[x=0.9cm,y=0.9cm]
\draw[black!35,-stealth] (-2.2,0) -- (2.4,0) node[below,font=\scriptsize] {$z_1$};
\draw[black!35,-stealth] (0,-2.2) -- (0,2.4) node[left,font=\scriptsize] {$z_2$};
\draw[thick,black!70] (-1.2,2.4) -- (2.4,-0.6) node[below,font=\scriptsize] {$Az=y$};
\draw[fa,thick,fill=fa!12] (1.4,0) -- (0,1.4) -- (-1.4,0) -- (0,-1.4) -- cycle;
\draw[fb,thick] (0,0) circle (1.076);
\fill[fa] (0,1.4) circle (2.2pt) node[right=2pt,font=\scriptsize,text=fa] {$\ell_1$: sparse};
\fill[fb] (0.69,0.83) circle (2.2pt) node[below right=0pt,font=\scriptsize,text=fb] {$\ell_2$: dense};
\end{tikzpicture}
\end{column}
\end{columns}

\source{candes2004,donoho2006}

\note{
\begin{itemize}
\item Once we know the dictionary, there is a second question: for a given observation, which features are active and how strongly?
\item This is an underdetermined system: more unknowns than equations, so infinitely many solutions
\item Key insight: if we additionally require the solution to be sparse, it becomes unique
\item We can find it by minimizing the $\ell_1$ norm, which is a convex problem
\item Why $\ell_1$ and not $\ell_2$? The $\ell_1$ ball has sharp corners on the axes, so the optimizer lands where most coordinates are zero
\item The $\ell_2$ ball is round, so $\ell_2$ minimization spreads energy across all coordinates
\item This is exactly what the sparsity penalty in SAEs does
\item Compressed sensing tells us that under the right conditions, the sparse decomposition is unique
\end{itemize}
}
\end{frame}

\begin{frame}{Trouble with SAEs: Feature Absorption}
\begin{center}
\includegraphics[width=0.9\textwidth,height=0.75\textheight,keepaspectratio]{figures/mcdougall2024/absorption.png}
\end{center}

\source{mcdougall2024, chanin2024}

\note{
\begin{itemize}
\item SAEs will, if they can, make feature descriptions more sparse than they ``really are''
\item Example: because ``short'' starts with S, the SAE does not fire the ``starts with S'' feature on ``short''
\item When features co-occur, the SAE absorbs one into the other
\end{itemize}
}
\end{frame}

\begin{frame}{Trouble with SAEs: Feature Shrinkage}
\paperfig{figures/rajamanoharan2024/fig4.png}{03_transformer_features/figures/rajamanoharan2024/fig4-caption.txt}

\source{rajamanoharan2024}

\note{
\begin{itemize}
\item Since SAEs have to satisfy both the L1 sparsity and the L2 reconstruction loss, they settle where both gradients cancel
\item The L1 loss pushes to make the feature smaller than it actually is
\item This leads to systematic underestimation of feature strength
\end{itemize}
}
\end{frame}

\begin{frame}{Gated SAEs}
\paperfig{figures/rajamanoharan2024/fig5.png}{03_transformer_features/figures/rajamanoharan2024/fig5-caption.txt}

\source{rajamanoharan2024}

\note{
\begin{itemize}
\item A countermeasure for shrinkage: the SAE learns two different things separately
\item Whether the feature is on or off (gating path)
\item What the magnitude of the feature is (magnitude path)
\end{itemize}
}
\end{frame}

\begin{frame}{JumpReLU SAEs}
\[ a = \text{JumpReLU}_\theta(W_{\text{enc}} x) = W_{\text{enc}} x \cdot \mathbf{1}[W_{\text{enc}} x > \theta] \]

\vspace{0.5em}
\[ \mathcal{L} = \underbrace{\| x - \hat{x} \|_2^2}_{\text{reconstruction}} + \lambda \sum_i \underbrace{\tanh(c \cdot \| W_{\text{dec},i} \| \cdot a_i)}_{\approx\, \mathbf{1}[a_i > 0] \text{ for large } c} \]

\vspace{0.5em}
Hard threshold $\theta$: features are either fully on or off.
Sparsity penalty approximates L0, no shrinkage on active features.

\source{ameisen2025}

\note{
\begin{itemize}
\item Another fix for shrinkage. This tanh sparsity penalty is the variant Anthropic uses in its cross-layer transcoders
\item Hard threshold $\theta$: if pre-activation is above $\theta$, it passes through at true value; if below, hard-zeroed
\item The sparsity penalty $\tanh(c \cdot \|w\| \cdot a)$ approximates counting active features (L0 proxy) for large $c$
\item Once a feature is clearly active, $\tanh$ saturates to 1 and gradient vanishes , no pressure to shrink magnitude
\item Training dynamics: reconstruction loss wants $\theta$ low, sparsity loss wants $\theta$ high; equilibrium determines sparsity
\item Used in cross-layer transcoders for Anthropic's circuit tracing work
\end{itemize}
}
\end{frame}

\begin{frame}{Top-K SAEs}
\[ f_{\text{TopK}}(x) = \text{TopK}(W_{\text{enc}}(x - b_{\text{dec}})) \]

\vspace{0.5em}
Set all but the top $K$ feature activations to zero.
No L1 penalty needed.

\source{bussmann2024}

\note{
\begin{itemize}
\item As often in ML, the simpler solution works more robustly
\item Completely get rid of the L1 norm, just set all but the top K feature activations to zero (Top-K SAEs: Gao et al., 2024)
\item BatchTopK (cited here): pick the top $K \times \text{batch size}$ activations across the whole batch, so the number of active features can vary per token
\item There has been a Cambrian explosion of SAE variants
\end{itemize}
}
\end{frame}

\begin{frame}{Matryoshka SAEs}
\paperfig{figures/bussmann2025/fig0.png}{03_transformer_features/figures/bussmann2025/fig0-caption.txt}

\source{bussmann2025}

\note{
\begin{itemize}
\item One dictionary, trained so that nested prefixes of it (the first $m_1$ latents, the first $m_2$, \ldots) each have to reconstruct the activation on their own
\item Early latents are forced to capture general concepts (``animal''), later latents add specifics (``dog'', ``cat'')
\item Reduces absorption: the general latent must also fire on the specific cases, because small prefixes cannot use the later latents
\end{itemize}
}
\end{frame}

\begin{frame}{Comparing SAEs}
\paperfig{figures/karvonen2025b/fig1.png}{03_transformer_features/figures/karvonen2025b/fig1-caption.txt}

\source{karvonen2025b}

\note{
\begin{itemize}
\item SAEs are not classical ML where we just want low loss on an eval set
\item We actually want them to be a good interpretability tool, and that is hard to quantify
\item Loss recovered: what percentage of the transformer performance is recovered when we replace activations with the SAE reconstruction vs.\ delete the activations
\item Auto interpretability: give an LLM examples of when the feature was active/not active, it proposes an explanation, then predicts feature activation on a test set
\item Mean absorption score: for hierarchical features like starting letters and words, how much are rarer features absorbed into more common ones?
\item SCR top 20: train a classification probe on data with spurious correlations (e.g.\ profession and gender), then ablate the gender feature --- does the bias decrease?
\item K-sparse probing: take a concept like sentiment, train a probe on the single most relevant feature --- what is the probe's performance?
\item Extremely mixed results: different SAE architectures have different strengths and weaknesses
\end{itemize}
}
\end{frame}

\begin{frame}{Transcoders and Crosscoders}
\begin{center}
\includegraphics[width=0.9\textwidth,height=0.75\textheight,keepaspectratio]{figures/lindsey2024/cross_trans_coer.png}
\end{center}

\source{lindsey2024}

\note{
\begin{itemize}
\item We can not only vary the architecture and loss of the SAE itself, but also what kind of data we are acting on
\item Transcoders do not sparsify a single layer of activations, but reconstruct the next layer from the current layer
\item The idea: sparsify the computation going on in some part of the transformer
\item Crosscoders take multiple layers and reconstruct multiple layers out of multiple layers
\end{itemize}
}
