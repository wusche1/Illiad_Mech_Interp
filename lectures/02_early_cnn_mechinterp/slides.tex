\begin{frame}{Seven Lines of Evidence for Curve Detectors}
\centering
\evidence{figures/curve_detector_feature_visualization.png}{1. Feature visualization}{optimized inputs are curves}\hfill
\evidence{figures/curve_detector_dataset_examples.png}{2. Dataset examples}{top images contain curves}\hfill
\evidence{figures/curve_detector_synthetic_tuning.png}{3. Synthetic examples}{fires on drawn curves, not lines}\hfill
\evidence{figures/curve_detector_radial_tuning.png}{4. Joint tuning}{rotation hands off to the next detector}

\vspace{0.2em}
\evidence{figures/curve_detector_weights.png}{5. Implementation}{curve algorithm readable from weights}\hspace{1.5em}
\evidence{figures/curve_detector_use_in_circuits.png}{6. Use}{downstream: circles, spirals}\hspace{1.5em}
\evidence{figures/arg_hand.png}{7. Handwritten circuit}{hand-set weights mimic the original}

\source{olah2020}

\note{
\begin{itemize}
\item 1--4 are behavioral: what makes the neuron fire
\item Feature visualization establishes a causal link: everything in the image was added to make the neuron fire
\item Dataset examples: strongest activations are curves in the expected orientation, weaker ones are imperfect curves
\item Synthetic curves: fires only near the expected orientation, not for straight lines or sharp corners
\item Joint tuning: rotating a stimulus smoothly transfers activation between orientation-selective detectors, which tile 360 degrees
\item 5--7 are circuit-based: how the neuron is built and used
\item Implementation: the curve detection algorithm can be read off the weights, with no sign of a second cause of firing
\item Use: downstream clients are features that naturally involve curves
\item Handwritten circuits: a cleanroom reimplementation with hand-set weights mimics the original
\item Compare this bar with what we accept for SAE features in language models later
\end{itemize}
}
\end{frame}

\begin{frame}{Universality in Image Models}
\begin{center}
\includegraphics[width=0.95\textwidth,height=0.8\textheight,keepaspectratio]{figures/universality.png}
\end{center}

\source{olah2020}

\note{
\begin{itemize}
\item In image models, these features at least in early layers seem to be pretty universal
\item Curve detectors and high-low frequency detectors appear across AlexNet, GoogLeNet, VGG19, and ResNet
\item Later layers often correspond to human concepts, which speaks for their universality
\item But many channels do not seem to have easily understandable descriptions
\end{itemize}
}
\end{frame}

\begin{frame}{If Universality Is True}
\begin{itemize}
\item What features are universal? Why?
\item Discussed in the Natural Abstractions discourse
\end{itemize}

\source{wentworth2020, chan2023}

\note{
\begin{itemize}
\item For curves this is obvious, but it is not clear how far this universality scales
\item Even if it does, it is not clear why those concepts in particular are universal
\item This might be pretty important: if we find a ``morally good'' feature in LLMs, do we expect it to behave the same as the human concept OOD in the limit of a super capable model?
\item Unclear, there are some theories here. AFAIK the work in this direction has stalled out the last years
\end{itemize}
}
